+        self.repo_root = self._find_repo_root(directory)
+
+    @staticmethod
+    def _find_repo_root(directory: Path):
+        try:
+            result = subprocess.run(
+                [
+                    "git",
+                    "-C",
+                    str(directory),
+                    "rev-parse",
+                    "--show-toplevel",
+                ],

+                text=True,

+            )
